import json, urllib.request, urllib.error, random, statistics, sys, os, time
# sweep.py <tag> <T list comma> [reps=3] : 真实文本,随机偏移(nonce),n_predict=1,cache_prompt=false,校验 prompt_n==T
B="http://127.0.0.1:9902"
def post(p,b,to=1800):
    return json.load(urllib.request.urlopen(urllib.request.Request(B+p,data=json.dumps(b).encode(),headers={"Content-Type":"application/json"}),timeout=to))
tag=sys.argv[1]; Ts=[int(x) for x in sys.argv[2].split(",")]; reps=int(sys.argv[3]) if len(sys.argv)>3 else 3
tf="~/notes/bench/e152/speed.tokens.q38.json"
if not os.path.exists(tf):
    toks=post("/tokenize",{"content":open("~/notes/bench/e150/corpus-speed.txt").read()})["tokens"]
    json.dump(toks,open(tf,"w"))
toks=json.load(open(tf)); rng=random.Random(hash(tag)&0xffff)
out=open(f"~/notes/bench/e407/sweep-{tag}.jsonl","a")
for T in Ts:
    ms=[]
    for r in range(reps+1):
        for att in range(5):
            o=rng.randrange(0,len(toks)-T-1)
            try:
                d=post("/completion",{"prompt":toks[o:o+T],"n_predict":1,"cache_prompt":False,"temperature":0}); break
            except urllib.error.HTTPError as e:
                print(f"  retry T={T} HTTP {e.code} (单 token 半个 UTF-8 字节解析失败)",flush=True)
        t=d["timings"]; assert t["prompt_n"]==T, (T,t["prompt_n"])
        if r>0: ms.append(t["prompt_ms"])
    med=statistics.median(ms)
    rec={"tag":tag,"T":T,"ms":ms,"med_ms":med,"tps":T/med*1000}
    out.write(json.dumps(rec)+"\n"); out.flush()
    print(f"{tag} T={T:5d} ms={' '.join(f'{x:.0f}' for x in ms)} med={med:.0f} -> {rec['tps']:.1f} t/s",flush=True)
