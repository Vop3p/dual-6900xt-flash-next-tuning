import json, urllib.request, statistics, sys
# short.py <tag>: E147 同题,1 warmup + 3 中位,记接受率
B="http://127.0.0.1:9902"; tag=sys.argv[1]
P=[("散文","请写一段约 250 字的说明文，主题是潮汐的成因。"),("代码","请用 Python 写一个快速排序函数，并加注释。")]
for lab,p in P:
    tg=[];acc=[]
    for i in range(4):
        b={"messages":[{"role":"user","content":p}],"max_tokens":256,"cache_prompt":False,
           "chat_template_kwargs":{"enable_thinking":False},"temperature":1,"top_k":20,"top_p":0.95}
        t=json.load(urllib.request.urlopen(urllib.request.Request(B+"/v1/chat/completions",data=json.dumps(b).encode(),headers={"Content-Type":"application/json"}),timeout=900))["timings"]
        if i: tg.append(t["predicted_per_second"]); acc.append(t.get("draft_n_accepted",0)/max(1,t.get("draft_n",1)))
    print(f"{tag} {lab}: {' '.join(f'{x:.2f}' for x in tg)} 中位 {statistics.median(tg):.2f} | accept {' '.join(f'{a:.2f}' for a in acc)}",flush=True)
