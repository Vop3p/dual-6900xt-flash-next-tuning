# Where the time goes, stock Strata 0.1.43.1 vs the tuned production setup (E449): two stacked-bar charts
# (prompt, decode), en/zh x light/dark.  Reads data.json from parse.py.  Run with a Python that has playwright.
import asyncio, json, os, statistics as st, sys

HERE = os.path.dirname(os.path.abspath(__file__))
OUT = sys.argv[1] if len(sys.argv) > 1 else os.path.join(HERE, 'png')
D = json.load(open(os.path.join(HERE, 'data.json')))
med = st.median

# ---------------------------------------------------------------- data
PC = ['hc read', 'GDN', 'QSA proj', 'QSA select+attn', 'experts', 'other']
PC_ZH = ['hc 读', 'GDN', 'QSA 投影', 'QSA 选块+注意力', '专家', '其他']
PC_EN = ['hc read', 'GDN', 'QSA projections', 'QSA select + attention', 'experts', 'other']
DC = ['hc read', 'dense GEMV', 'attn + GDN rec', 'routed experts', 'waits', 'draft + commit']
DC_ZH = ['hc 读 + 路由', '稠密 GEMV（投影、共享专家、head）', '注意力 + GDN 递推', '路由专家', '等待（CPU 专家、PCIe）', '草稿 + 提交']
DC_EN = ['hc read + router', 'dense GEMV (projections, shared expert, head)', 'attention + GDN recurrence', 'routed experts',
         'waits (CPU experts, PCIe)', 'draft + commit']


def prompt_rows():
    rows = []
    for t in ('4096', '32768', '128000'):
        r = {}
        for side, arms in (('s', ('s-t1', 's-t2')), ('p', ('p-t1', 'p-t2'))):
            reqs = [q for a in arms if a in D for q in D[a]['prompt'][t]]
            ms1k = med(q['_ttft'] / q['_tokens'] * 1000 for q in reqs)
            tot = {c: sum(q[c] for q in reqs) for c in PC}
            s = sum(tot.values())
            r[side] = dict(total=ms1k, tok_s=1e6 / ms1k, seg=[tot[c] / s * ms1k for c in PC],
                           share=[tot[c] / s for c in PC], n=len(reqs))
        rows.append((t, r))
    return rows


def decode_rows():
    rows = []
    for i, t in enumerate(('4096', '32768', '128000')):
        r = {}
        for side, tarms, varm in (('s', ('s-t1', 's-t2'), 's-v'), ('p', ('p-t1', 'p-t2'), 'p-v')):
            # t arms: requests in order 4K, 4K, 32K, 32K, 128K, 128K
            ms = [D[a]['decode'][2 * i + k]['ms_tok'] for a in tarms if a in D for k in (0, 1)]
            tot_ms = med(ms)
            stg = D[varm]['stages'][2 * i:2 * i + 2] if i < 2 else D[varm + '128']['stages'][0:2]
            tot = {c: sum(q.get(c, 0) for q in stg) for c in DC}
            s = sum(tot.values())
            r[side] = dict(total=tot_ms, tok_s=1000 / tot_ms, seg=[tot[c] / s * tot_ms for c in DC],
                           share=[tot[c] / s for c in DC], n=len(ms))
        rows.append((t, r))
    return rows

def stack(seg, total, mx, unit, speed, fmt, k=0.70):
    w = total / mx * 100 * k
    cells = ''
    for j, v in enumerate(seg):
        pct = v / total * 100
        lab = fmt(v) if v / mx * 1100 * k * 0.86 > 44 else ''   # label only where it fits (~44 px)
        cells += f'<i style="--c:var(--c{j + 1});flex:{v:.4f} 0 0"><em>{lab}</em></i>'
    return (f'<div class="track"><div class="stack" style="width:{w:.2f}%">{cells}</div>'
            f'<span class="tot" style="left:{w + 1:.2f}%"><b>{fmt(total)}</b> {unit} · {speed}</span></div>')


def page(kind, lang, theme):
    zh = lang == 'zh'
    if kind == 'prompt':
        rows, names = prompt_rows(), (PC_ZH if zh else PC_EN)
        mx = max(r[s]['total'] for _, r in rows for s in 'sp')
        unit = '毫秒 / 千 token' if zh else 'ms per 1K tokens'
        fmt = lambda v: f'{v:,.0f}'
        sp = lambda d: f'{d["tok_s"]:,.0f} tok/s'
        ctxn = {'4096': '4K', '32768': '32K', '128000': '128K'}
        h = ('读 prompt 的时间花在哪：原版 0.1.43.1 对比调优后（E449）' if zh else
             'Where reading a prompt goes: stock Strata 0.1.43.1 against the tuned setup (E449)')
        sub = ('柱长 = 每读 1,000 个 token 的毫秒数（客户端首 token 时间）；越短越快' if zh else
               'Bar length = milliseconds per 1,000 prompt tokens (client time to first token); shorter is faster')
    else:
        rows, names = decode_rows(), (DC_ZH if zh else DC_EN)
        mx = max(r[s]['total'] for _, r in rows for s in 'sp')
        unit = '毫秒 / token' if zh else 'ms per token'
        fmt = lambda v: f'{v:.1f}'
        sp = lambda d: f'{d["tok_s"]:.1f} tok/s'
        ctxn = {'4096': '4K 上下文' if zh else '4K context', '32768': '32K 上下文' if zh else '32K context',
                '128000': '128K 上下文' if zh else '128K context'}
        h = ('解码的时间花在哪：原版 0.1.43.1 对比调优后（E449）' if zh else
             'Where decoding goes: stock Strata 0.1.43.1 against the tuned setup (E449)')
        sub = ('柱长 = 每输出 1 个 token 的毫秒数；越短越快' if zh else
               'Bar length = milliseconds per output token; shorter is faster')
    body = ''
    for t, r in rows:
        x = r['s']['total'] / r['p']['total']
        lab_s, lab_p = ('原版', '调优') if zh else ('stock', 'tuned')
        body += (f'<div class="grp"><div class="ctx">{ctxn[t]}<br><span class="who">×{x:.2f}</span></div>'
                 f'<div class="row"><span class="who">{lab_s}</span>{stack(r["s"]["seg"], r["s"]["total"], mx, unit, sp(r["s"]), fmt)}</div>'
                 f'<div class="row"><span class="who">{lab_p}</span>{stack(r["p"]["seg"], r["p"]["total"], mx, unit, sp(r["p"]), fmt)}</div></div>')
    leg = '<div class="legend">' + ''.join(f'<span style="--c:var(--c{j + 1})">{n}</span>' for j, n in enumerate(names)) + '</div>'
    note = NOTES[(kind, lang)]
    return (f'<!doctype html><html lang="{lang}" data-theme="{theme}"><head><meta charset="utf-8">'
            '<link rel="stylesheet" href="https://fonts.googleapis.com/css2?family=Noto+Sans+SC:wght@400;500&family='
            'Noto+Serif+SC:wght@600&family=IBM+Plex+Mono:wght@400;600&display=swap">'
            f'<style>{CSS}</style></head><body><div class="fig" id="f"><h2>{h}</h2><p class="sub">{sub}</p>{leg}'
            f'<div class="panel">{body}</div><div class="note">{note}</div></div></body></html>')

async def main():
    from playwright.async_api import async_playwright
    os.makedirs(OUT, exist_ok=True)
    async with async_playwright() as p:
        b = await p.chromium.launch()
        pg = await (await b.new_context(viewport={"width": 1100, "height": 900}, device_scale_factor=2)).new_page()
        for kind in ('prompt', 'decode'):
            for lang in ('en', 'zh'):
                for theme in ('light', 'dark'):
                    f = os.path.join(OUT, f'{kind}-{lang}-{theme}.html')
                    open(f, 'w').write(page(kind, lang, theme))
                    await pg.goto('file://' + f, wait_until='networkidle')
                    await pg.evaluate('document.fonts.ready'); await pg.wait_for_timeout(300)
                    await (await pg.query_selector('#f')).screenshot(path=os.path.join(OUT, f'breakdown-{kind}-{lang}-{theme}.png'))
                    print('wrote', kind, lang, theme)
        await b.close()


if __name__ == '__main__':
    asyncio.run(main())
