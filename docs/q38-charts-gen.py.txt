def fmt(v):
    return f"{v:,.0f}" if v >= 100 else f"{v:g}"

def speed_svg(lang):
    # steps: (label line1, label line2, p2k, p8k, p32k, p128k, decode)
    steps = [
        ("docker image", "09-26", 617, 623, 566, None, None),
        ("host + MMQ tiles", "09-29", 683, 677, 600, None, None),
        ("mad24", "10-02", 774, 773, 674, None, None),
        ("q8k ggml", "10-08", 812, 811, 705, 460.5, 62.9),
        ("int8 QK FA", "10-09", 814, 820, 732, 511.6, 62.3),
    ]
    W, H = 1070, 520; L, R, T, B = 70, 70, 40, 90
    n = len(steps); xs = [L + i * (W - L - R) / (n - 1) for i in range(n)]
    ymax, dmax = 900, 90
    def y(v): return T + (H - T - B) * (1 - v / ymax)
    def yd(v): return T + (H - T - B) * (1 - v / dmax)
    s = [f'<svg class="chart" viewBox="0 0 {W} {H}" width="{W}" height="{H}">']
    for g in range(0, ymax + 1, 150):
        s.append(f'<line x1="{L}" x2="{W-R}" y1="{y(g):.1f}" y2="{y(g):.1f}" stroke="var(--rule)"/>')
        s.append(f'<text x="{L-8}" y="{y(g)+4:.1f}" text-anchor="end">{g:,}</text>')
    for g in range(0, dmax + 1, 30):
        s.append(f'<text x="{W-R+8}" y="{yd(g)+4:.1f}">{g}</text>')
    s.append(f'<text x="{L}" y="{T-16}">tok/s</text><text x="{W-R+8}" y="{T-16}">{"解码" if lang=="zh" else "decode"}</text>')
    series = [(2, "--c1", False), (3, "--c2", False), (4, "--c3", False), (5, "--c4", False), (6, "--c5", True)]
    # (index, color, right axis)
    series = [(2, "--c1", False), (4, "--c3", False), (5, "--c5", False), (6, "--c4", True)]
    for idx, col, right in series:
        pts = [(xs[i], (yd if right else y)(st[idx]), st[idx]) for i, st in enumerate(steps) if st[idx] is not None]
        for (x0, y0, _), (x1, y1, _) in zip(pts, pts[1:]):
            s.append(f'<line x1="{x0:.1f}" y1="{y0:.1f}" x2="{x1:.1f}" y2="{y1:.1f}" stroke="var({col})" stroke-width="3"/>')
        for x0, y0, v in pts:
            s.append(f'<circle cx="{x0:.1f}" cy="{y0:.1f}" r="5.5" fill="var({col})"/>')
            dy = -10 if idx in (2, 4) else 20
            if right and x0 == pts[-1][0]:
                s.append(f'<text x="{x0-9:.1f}" y="{y0+dy:.1f}" text-anchor="end" style="fill:var({col})">{fmt(v)}</text>')
            else:
                s.append(f'<text x="{x0+8:.1f}" y="{y0+dy:.1f}" style="fill:var({col})">{fmt(v)}</text>')
    for i, st in enumerate(steps):
        yy = H - B + 22 + (0 if i % 2 == 0 else 34)
        s.append(f'<text x="{xs[i]:.1f}" y="{yy}" text-anchor="middle">{st[0]}</text><text x="{xs[i]:.1f}" y="{yy+16}" text-anchor="middle">{st[1]}</text>')
    s.append('</svg>')
    return "\n".join(s)

async def main():
    from playwright.async_api import async_playwright
    async with async_playwright() as p:
        b = await p.chromium.launch()
        ctx = await b.new_context(viewport={"width": 1100, "height": 900}, device_scale_factor=2)
        pg = await ctx.new_page()
        for lang in ("en", "zh"):
            for theme in ("light", "dark"):
                path = os.path.join(HERE, f"{lang}-{theme}.html")
                open(path, "w").write(page(lang, theme))
                await pg.goto("file://" + path, wait_until="networkidle")
                await pg.evaluate("document.fonts.ready")
                await pg.wait_for_timeout(300)
                for fig in ("now", "speed", "decode"):
                    el = await pg.query_selector(f"#{fig}")
                    await el.screenshot(path=os.path.join(OUT, f"q38-{fig}-{lang}-{theme}.png"))
                    print("wrote", f"q38-{fig}-{lang}-{theme}.png")
        await b.close()
asyncio.run(main())
